\item Calculate the entropy of a fair six-sided dice.

$M = \{1,2,3,4,5,6\}$

$P(M_i) = \frac{1}{6}$

$H = - \sum_{i=1}^{M}p_i\log_2(p_i) = -6 * \frac{1}{6} * \log_2\frac{1}{6} = -\log_2\frac{1}{6} \approx 2.585$ bits


\item The link contains an excerpt from the novella «The Little Prince». Calculate its entropy, considering only the frequency of letters of the English alphabet.\\

\begin{verbatim}
from math import log2
def find_entropy(s):
    allowed_symbols = set(list('ABCDEFGHIJKLMNOPQRSTUVWXYZ'))
    s = s.upper()
    s = [i for i in s if i in allowed_symbols]
    hashmap_p = {i: s.count(i)/len(s) for i in list(allowed_symbols)}
    H = 0
    for i in hashmap_p:
        H += hashmap_p[i] * log2(hashmap_p[i]+10**-10)
    H *= -1
    return(round(H,3))
\end{verbatim}
